\newcommand{\setrootdepth}[1]{
    \def\rootpath{}
    \count0=0
    \loop
        \ifnum\count0<#1
            \edef\rootpath{\rootpath../}
            \advance\count0 by 1
    \repeat
 
    \graphicspath{{\rootpath fig/}{\rootpath uml/}} % Change this to be a relative path to your figures.
    \addbibresource{\rootpath bib/references.bib} % Change this to be a relative path to your references.bib file.
    \def\input@path{{\rootpath inc/}} % Change this to be a relative path to your included tex files.
}
